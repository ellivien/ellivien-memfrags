Cloudflare Worker memory-fragments route addition
=================================================

Merge this route into the existing Ellivien Worker's authenticated `/api/*`
request handler. It assumes that the Worker already supplies `url`,
`env.PORTAL_STORAGE` and `jsonResponse(...)`.

This is not a complete Worker and does not create a Worker, namespace, binding
or credential.

if (url.pathname === '/api/memory-fragments') {
  function normaliseMemoryFragments(rawFragments) {
    if (!Array.isArray(rawFragments)) {
      throw new Error('Memory fragments request must contain a fragments array.');
    }
    const seenTriggers = new Set();
    return rawFragments.map((fragment, index) => {
      if (!fragment || typeof fragment !== 'object' || Array.isArray(fragment)) {
        throw new Error(`Memory fragment ${index + 1} must be an object.`);
      }
      if (fragment.category != null && typeof fragment.category !== 'string') {
        throw new Error(`Memory fragment ${index + 1} category must be a string.`);
      }
      if (typeof fragment.definition !== 'string') {
        throw new Error(`Memory fragment ${index + 1} definition must be a string.`);
      }
      if (!Array.isArray(fragment.triggers) ||
          !fragment.triggers.every(trigger => typeof trigger === 'string')) {
        throw new Error(`Memory fragment ${index + 1} triggers must be an array of strings.`);
      }

      const category = (fragment.category || '').trim();
      const definition = fragment.definition.trim();
      const triggers = [];
      const rowTriggerKeys = new Set();
      fragment.triggers
        .map(trigger => trigger.trim())
        .filter(Boolean)
        .forEach(trigger => {
          const key = `${category.toLowerCase()}::${trigger.toLowerCase()}`;
          if (rowTriggerKeys.has(key)) return;
          rowTriggerKeys.add(key);
          triggers.push(trigger);
        });

      if (!definition || triggers.length === 0) {
        throw new Error(`Memory fragment ${index + 1} needs at least one trigger and a definition.`);
      }

      triggers.forEach(trigger => {
        const key = `${category.toLowerCase()}::${trigger.toLowerCase()}`;
        if (seenTriggers.has(key)) {
          const categoryLabel = category ? ` in category "${category}"` : '';
          throw new Error(`Duplicate memory-fragment trigger: ${trigger}${categoryLabel}`);
        }
        seenTriggers.add(key);
      });

      return { category, triggers, definition };
    });
  }

  if (request.method === 'GET') {
    let stored;
    try {
      stored = await env.PORTAL_STORAGE.get('memory-fragments');
    } catch {
      return jsonResponse({ error: 'Stored memory fragments could not be read.' }, 500);
    }
    if (stored === null) return jsonResponse({ fragments: [] });
    try {
      const parsed = JSON.parse(stored);
      return jsonResponse({ fragments: normaliseMemoryFragments(parsed?.fragments) });
    } catch {
      return jsonResponse({ error: 'Stored memory fragments are invalid.' }, 500);
    }
  }

  if (request.method === 'POST') {
    let payload;
    try {
      payload = await request.json();
    } catch {
      return jsonResponse({ error: 'Memory fragments request must contain valid JSON.' }, 400);
    }
    let fragments;
    try {
      fragments = normaliseMemoryFragments(payload?.fragments);
    } catch (error) {
      return jsonResponse({ error: error.message || 'Memory fragments are invalid.' }, 400);
    }

    try {
      await env.PORTAL_STORAGE.put('memory-fragments', JSON.stringify({ fragments }));
    } catch {
      return jsonResponse({ error: 'Memory fragments could not be saved.' }, 500);
    }
    return jsonResponse({ success: true, count: fragments.length });
  }
}
